// ===== function _condenseAiSummary =====
function _condenseAiSummary(raw){
  let t = String(raw||'').replace(/\s+/g,' ').trim();
  if(!t) return '';
  t = t.replace(/^sentiment[^.!?]*[.!?]\s*/i, '');
  const sentences = t.match(/[^.!?]+[.!?]+/g) || [t];
  let out = sentences.slice(0, 2).map(function(s){ return s.trim(); }).join(' ').trim();
  if(out.length > 300) out = out.slice(0, 297).replace(/\s+\S*$/,'').trim() + '…';
  return out;
}

// ===== function importCohostFromAirDNA =====
function importCohostFromAirDNA(){
  const input = document.getElementById('cohost_airdna_url');
  const statusEl = document.getElementById('cohost_airdna_status');
  const url = (input && input.value || '').trim();
  if(!url){ if(statusEl) statusEl.textContent='Please paste an AirDNA link.'; return; }
  const isAirdna = /^https?:\/\/(?:[a-z0-9-]+\.)?airdna\.co\//i.test(url);
  if(!isAirdna){ if(statusEl){ statusEl.style.color='var(--danger)'; statusEl.textContent='Co-Hosting imports accept AirDNA links only.'; } return; }
  if(typeof rrScrapeViaExtension !== 'function'){ if(statusEl) statusEl.textContent='Extension unavailable.'; return; }
  if(input) input.disabled = true;
  if(statusEl){ statusEl.style.color='var(--text3)'; statusEl.textContent='Importing from AirDNA (pulling amenities from the listing — this can take a moment)…'; }
  // Prominent loading toast with progress bar — same indicator the Arbitrage
  // listing import uses (importFromChoicePaste). Replaced by the success/error
  // toast below when the scrape resolves.
  showToast('Importing from AirDNA…', 'loading', 30000);
  rrScrapeViaExtension(url, 'cohost').then(function(res){
    if(input) input.disabled = false;
    if(!res || !res.ok){
      const err = res && res.error;
      let msg = 'Could not import from AirDNA: ' + (err||'Unknown error');
      if(err==='extension-not-installed') msg = 'Install the RentingRadar Chrome extension to import from AirDNA.';
      else if(err==='airdna-login-required') msg = (res.detail||'Please log into AirDNA in your browser first, then retry.');
      else if(err==='airdna-paywall') msg = (res.detail||'This AirDNA listing requires a paid subscription.');
      if(statusEl){ statusEl.style.color='var(--danger)'; statusEl.textContent = msg; }
      showToast(msg, 'error', 6000);
      return;
    }
    const d = res.data || {};
    // Prefer the detail-panel-SCOPED co* financials (correct for the target
    // listing); fall back to the generic fields only if scoping missed.
    const annual = _cn(d.coAnnualRevenue || d.annualRevenue);
    // The property NAME is the LISTING TITLE (e.g. "Panoramic Mountain Views",
    // the host's listing name — confirmable on the Airbnb listing). NOT an
    // address: AirDNA STR listings have no street address.
    const title = (d.listingTitle || d.title || '').trim() || 'Untitled Listing';
    const r = {
      id: uid(), kind: 'cohost', status: 'New',
      property: title || 'Untitled Listing',
      links: { airbnb: d.link || '', vrbo: d.vrboLink || '', booking: d.bookingLink || '', maps: '' },
      cohost: {
        hostName: d.host || '',
        setupInvestment: 3000,
        cleaningPerMonth: 900,
        expectedRevenue: annual ? Math.round(annual/12) : '',
        expectedRevenueLow: '', expectedRevenueHigh: '',
        askingRent: '', feePct: '',
        airdnaUrl: d.airdnaUrl || url,
        annualRevenue: annual || '',
        occupancy: _cn(d.coOccupancy!=null?d.coOccupancy:d.occupancy)||'',
        nightlyRate: _cn(d.coADR!=null?d.coADR:d.nightlyRate)||'',
        daysAvailable: _cn(d.coDaysAvailable!=null?d.coDaysAvailable:d.daysAvailable)||'',
        revenuePotential: _cn(d.coRevenuePotential)||'', marketScore: (d.marketScore!=null?d.marketScore:''), priceTier: d.priceTier||'',
        amenities: Array.isArray(d._amenities) ? d._amenities : [],
        inconsistencies: Array.isArray(d._inconsistencies) ? d._inconsistencies : []
      },
      propertyDetails: Object.assign({}, d.propertyDetails||{}, {
        bedrooms: (d.bedrooms!=null?d.bedrooms:''), bathrooms: (d.bathrooms!=null?d.bathrooms:''),
        guests: (d.guests!=null?d.guests:''), beds: (d.beds!=null?d.beds:''),
        rating: (d.coRating!=null?d.coRating:(d.rating!=null?d.rating:'')),
        propertyType: d.airdnaType || (d.propertyDetails&&d.propertyDetails.propertyType) || '',
        market: d.market||'', submarket: d.submarket||'',
        listingTitle: title,
        amenitiesText: (d.description||'')
      }),
      details: { comps: [] },
      notes: [], tasks: [], meetings: [],
      createdAt: new Date().toISOString()
    };
    // Maps link: anchor on the listing's coordinates (or the reverse-geocoded
    // address, or the market) — AirDNA listings have no exact street address.
    if(!r.links.maps){
      const la=(d.coLat!=null?d.coLat:d.lat), ln=(d.coLng!=null?d.coLng:d.lng);
      if(la!=null && ln!=null) r.links.maps = 'https://www.google.com/maps/search/?api=1&query='+la+','+ln;
      else if(d.market||d.submarket) r.links.maps = 'https://www.google.com/maps/search/?api=1&query='+encodeURIComponent([d.submarket,d.market].filter(Boolean).join(', '));
    }
    // Seed the Property Note with a SUCCINCT overview of AirDNA's AI Summary —
    // never the verbatim blurb. Prefer the extension's Claude-condensed version
    // (d.aiSummaryShort); if that's unavailable (AI offline), fall back to a
    // local heuristic condense (first couple of sentences, capped).
    // Store the FULL AirDNA AI summary so the note can render its sections (Sentiment,
    // What guests like, Areas to improve, Location Highlights). Falls back to the
    // condensed version only if the full text wasn't captured.
    const aiFull = (d.aiSummary && String(d.aiSummary).trim()) || (d.aiSummaryShort && String(d.aiSummaryShort).trim()) || '';
    if(aiFull && aiFull.length > 20){
      r.propertyNote = { text: aiFull, html: _cohostRenderAiSummary(aiFull), date: new Date().toISOString(), source:'AirDNA AI' };
    }
    cohostRecords.unshift(r);
    saveCohostData();
    closeAddModal();
    showToast('Imported from AirDNA. Review the details and financials.', 'success', 4000);
    openCohostProfile(r.id);
  }).catch(function(e){
    if(input) input.disabled = false;
    if(statusEl){ statusEl.style.color='var(--danger)'; statusEl.textContent='Import error: '+(e.message||e); }
    showToast('Import error: '+(e.message||e), 'error', 6000);
  });
}

// ===== function _cohostNoteToBullets =====
function _cohostNoteToBullets(text){
  let t = String(text || '').trim();
  if(!t) return '';
  let parts = t.split(/\n+/).map(function(s){ return s.replace(/^[\-•*\s]+/, '').trim(); }).filter(Boolean);
  if(parts.length <= 1){
    parts = (t.match(/[^.!?]+[.!?]+/g) || [t]).map(function(s){ return s.trim(); }).filter(Boolean);
  }
  return '<ul style="margin:2px 0 0;padding-left:18px">'+parts.map(function(p){ return '<li style="margin:1px 0">'+esc(p)+'</li>'; }).join('')+'</ul>';
}

// ===== function _cohostRenderAiSummary =====
function _cohostRenderAiSummary(raw){
  let t = String(raw || '').replace(/\r/g, '').trim();
  if(!t) return '';
  const sm = t.match(/sentiment[^0-9]{0,12}(\d{1,3})\s*\/\s*100/i);
  const sentiment = sm ? sm[1] : '';
  const defs = [
    { keys:['what guests like','what guests are saying','what guests love','guests like','positives'], icon:'👍', label:'What guests like' },
    { keys:['areas to improve','areas for improvement','areas of improvement','room for improvement','concerns','negatives'], icon:'⚠️', label:'Areas to improve' },
    { keys:['location highlights','location highlight','neighborhood highlights'], icon:'📍', label:'Location Highlights' }
  ];
  const low = t.toLowerCase();
  const found = [];
  defs.forEach(function(d){ for(let j=0;j<d.keys.length;j++){ const i = low.indexOf(d.keys[j]); if(i!==-1){ found.push({ def:d, idx:i, hdrLen:d.keys[j].length }); break; } } });
  found.sort(function(a,b){ return a.idx - b.idx; });
  if(found.length === 0) return _cohostNoteToBullets(t);
  let html = '';
  if(sentiment){
    html += '<div style="display:flex;align-items:center;gap:6px;margin-bottom:7px"><span style="display:inline-flex;align-items:center;gap:4px;font-size:.78rem;font-weight:700;background:rgba(58,210,159,.14);color:var(--success);border-radius:6px;padding:1px 8px">📊 Sentiment '+esc(sentiment)+'/100</span><span style="font-weight:400;font-size:.66rem;color:var(--text3)">AI-analyzed reviews</span></div>';
  }
  found.forEach(function(f, i){
    const start = f.idx + f.hdrLen;
    const end = (i+1 < found.length) ? found[i+1].idx : t.length;
    let body = t.slice(start, end).trim();
    let parts = body.split(/\n+/).map(function(s){ return s.replace(/^[\-•*·\s]+/, '').trim(); }).filter(function(s){ return s.length > 1; });
    if(parts.length <= 1){ parts = (body.match(/[^.!?]+[.!?]+/g) || [body]).map(function(s){ return s.trim(); }).filter(Boolean); }
    html += '<div style="margin-bottom:8px"><div style="display:flex;align-items:center;gap:6px;font-weight:700;font-size:.82rem;color:var(--text);margin-bottom:2px"><span style="font-size:.92rem">'+f.def.icon+'</span>'+esc(f.def.label)+'</div>';
    html += '<ul style="margin:0;padding-left:20px">'+parts.map(function(p){ return '<li style="margin:1px 0;font-size:.82rem;color:var(--text2)">'+esc(p)+'</li>'; }).join('')+'</ul></div>';
  });
  return html;
}

// ===== function _cohostMergeIncons =====
function _cohostMergeIncons(incons){
  const out = [], byField = {};
  (incons||[]).forEach(function(it){
    if(it && it.values){
      const key = String(it.field||'');
      if(byField[key]){
        Object.keys(it.values).forEach(function(k){ if(!(k in byField[key].values)) byField[key].values[k] = it.values[k]; });
        return;
      }
      byField[key] = { field: it.field, severity: it.severity, value: it.value, values: Object.assign({}, it.values) };
      out.push(byField[key]);
    } else { out.push(it); }
  });
  return out;
}

// ===== function _cohostIncRow =====
function _cohostIncRow(it, srcs, extraHeader){
  const conflict = it.severity === 'conflict';
  const col = conflict ? 'var(--danger)' : 'var(--warning)';
  const bg = conflict ? 'rgba(239,68,68,.06)' : 'rgba(245,158,11,.06)';
  const tag = conflict ? 'Conflict' : 'Mismatch';
  const label = esc(it.field||'Difference'); // field name only — value text dropped
  const _indicator = function(present, absent, src){
    const key = src==='AirBNB' ? 'Airbnb' : src; // VRBO→VRBO, B.com→B.com
    if((present||[]).indexOf(key)!==-1) return '<span style="color:var(--success);font-weight:700">✓</span>';
    if((absent||[]).indexOf(key)!==-1) return '<span style="color:var(--danger);font-weight:700">✗</span>';
    return '<span style="color:var(--text3)">—</span>';
  };
  let h = '<div style="border-left:3px solid '+col+';padding:8px 12px;background:'+bg+';border-radius:0 8px 8px 0">';
  h += '<div style="display:flex;align-items:center;gap:8px;margin-bottom:5px"><span style="font-weight:600;font-size:.82rem">'+label+'</span><span style="font-size:.58rem;font-weight:700;text-transform:uppercase;letter-spacing:.4px;color:'+col+';border:1px solid '+col+';border-radius:10px;padding:0 6px">'+tag+'</span>'+(extraHeader||'')+'</div>';
  if(it.values){
    h += '<div style="display:flex;flex-direction:column;gap:3px">';
    srcs.forEach(function(src){
      if(!(src in it.values) && !((src==='AirBNB'?'Airbnb':src) in it.values)) return;
      const v = it.values[src] || it.values[src==='AirBNB'?'Airbnb':src];
      h += '<div style="font-size:.78rem;display:flex;gap:8px"><span style="color:var(--text3);min-width:54px">'+src+'</span><span style="color:var(--text)">'+(v?esc(v):'—')+'</span></div>';
    });
    h += '</div>';
  } else {
    h += '<div style="display:flex;gap:26px;font-size:.8rem;align-items:center">'+
      srcs.map(function(src){ return '<span><span style="color:var(--text3);margin-right:7px">'+src+'</span>'+_indicator(it.present,it.absent,src)+'</span>'; }).join('')+
      '</div>';
  }
  h += '</div>';
  return h;
}

// ===== function cohostToggleIncShow =====
function cohostToggleIncShow(id, field, checked){
  const r = getCohostById(id); if(!r) return;
  if(!r.cohost) r.cohost = {};
  let arr = Array.isArray(r.cohost.shownInconsistencies) ? r.cohost.shownInconsistencies.slice() : [];
  const i = arr.indexOf(field);
  if(checked && i === -1) arr.push(field);
  else if(!checked && i !== -1) arr.splice(i, 1);
  r.cohost.shownInconsistencies = arr;
  _saveRecFor(r, id);
}

// ===== function _cohostSummaryInconsistencies =====
function _cohostSummaryInconsistencies(r){
  const c = r.cohost || {};
  const incons = Array.isArray(c.inconsistencies) ? c.inconsistencies : [];
  const shown = Array.isArray(c.shownInconsistencies) ? c.shownInconsistencies : [];
  if(!incons.length || !shown.length) return '';
  const L = r.links || {};
  const srcs = ['AirBNB','VRBO'].concat(L.booking ? ['B.com'] : []);
  const rows = _cohostMergeIncons(incons).filter(function(it){ return shown.indexOf(it.field) !== -1; });
  if(!rows.length) return '';
  let h = '<div style="background:var(--bg2);border:1px solid var(--border);border-radius:12px;padding:18px 20px;margin-top:14px">';
  h += '<div style="font-weight:600;margin-bottom:4px">⚠️ Listing Inconsistencies to Discuss</div>';
  h += '<div style="font-size:.76rem;color:var(--text3);margin-bottom:14px">Differences you flagged across the landlord’s listings — talking points to get them back in sync.</div>';
  h += '<div style="display:flex;flex-direction:column;gap:9px">';
  rows.forEach(function(it){ h += _cohostIncRow(it, srcs, ''); });
  h += '</div></div>';
  return h;
}
